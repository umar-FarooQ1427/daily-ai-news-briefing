name: Daily AI News Collection and Analysis

on:
  schedule:
    - cron: "0 3 * * *"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  collect-and-analyze:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: python -m pip install -r requirements.txt
      - run: python news_collector.py
      - name: Analyze news with Gemini AI
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          GEMINI_MODEL: "gemini-2.5-flash"
        run: python ai_news_analyzer.py
      - uses: actions/upload-artifact@v4
        with:
          name: daily-news-reports
          path: |
            news.json
            daily_briefing.md
            ai_briefing.md
